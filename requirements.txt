# STEP 1 - install PyTorch first, separately (pick ONE line):
#   Have an NVIDIA GPU:  pip install torch==2.5.1 torchvision==0.20.1 torchaudio==2.5.1 --index-url https://download.pytorch.org/whl/cu121
#   No NVIDIA GPU:       pip install torch==2.5.1 torchvision==0.20.1 torchaudio==2.5.1

# STEP 2 - install stable-baselines3 on its own, with --no-deps:
#   pip install stable-baselines3==2.9.0 --no-deps
#   (Why --no-deps: it asks for a torch version newer than any CUDA build
#   that exists, so without this flag it silently swaps out the torch you
